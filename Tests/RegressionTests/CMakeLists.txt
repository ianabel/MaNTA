# The regression suite: run the solver over every .conf here, compare against the
# checked-in .ref.nc references, then check the restart round trips.
#
# Run from this directory's *binary* dir. TestSolutions.py reads its .conf inputs
# and .ref.nc references from beside itself and writes everything else -- the
# runs' output and the config variants the restart checks generate -- into the
# cwd, so nothing it does lands in the source tree. SOLVER names the binary.
#
# Run through ${Python3_EXECUTABLE} rather than the script's own `env python3`
# shebang, so it gets the interpreter that has netCDF4, scipy and matplotlib
# rather than whichever python3 is first on PATH.
if(NOT Python3_EXECUTABLE)
  find_package(Python3 COMPONENTS Interpreter QUIET)
endif()

if(NOT Python3_EXECUTABLE)
  message(STATUS "No Python 3 interpreter found: the regression suite is not registered")
  return()
endif()

# run-regression is what CTest runs and what to run by hand, from any cwd; its
# arguments are the script's (--tolerance).
set(MANTA_RUN_REGRESSION "${CMAKE_CURRENT_BINARY_DIR}/run-regression")
file(GENERATE OUTPUT "${MANTA_RUN_REGRESSION}"
  CONTENT "#!/bin/sh
# Generated by CMake. The regression suite against the build at
# ${PROJECT_BINARY_DIR}, as CTest runs it.
cd '${CMAKE_CURRENT_BINARY_DIR}' || exit 1
export SOLVER='$<TARGET_FILE:MaNTA>'
exec '${Python3_EXECUTABLE}' '${CMAKE_CURRENT_SOURCE_DIR}/TestSolutions.py' \"$@\"
"
  FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE
                   WORLD_READ WORLD_EXECUTE)

add_test(NAME regression
         COMMAND "${MANTA_RUN_REGRESSION}"
         WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(regression PROPERTIES
  LABELS "regression"
  TIMEOUT 3600)

add_custom_target(regression_tests
  COMMAND "${CMAKE_CTEST_COMMAND}" --test-dir "${PROJECT_BINARY_DIR}" -R "^regression$" --output-on-failure
  USES_TERMINAL
  COMMENT "Running the regression suite")
add_dependencies(regression_tests MaNTA)
