# pytest over the manta package, run from the build tree.
#
#   cwd        conftest.py moves each test into a scratch directory under
#              --basetemp, so output a run drops lands in this binary dir
#   imports    the package this build assembled (PYTHONPATH, via
#              MANTA_PYTHON_ENV), and conftest.py refuses to start if `manta`
#              resolves anywhere else -- an installed copy, say
#   caches     pytest's under cache_dir, bytecode under PYTHONPYCACHEPREFIX; both
#              default to the source tree
#
# The tests' own inputs -- the .conf files, the .ref.nc references and the case
# modules they name -- are read from beside the test files. pytest.ini is named
# explicitly because rootdir discovery starts from the arguments and the cwd,
# neither of which need be inside the repository.
#
# All of that is in one generated launcher, run-pytest, which is what CTest runs
# and what to run by hand: its arguments are pytest's, so
#
#     <build>/python/Tests/run-pytest python/Tests/test_adjoint.py -k closed_form
#
# runs one test exactly as the suite would, from any cwd. With no arguments it
# runs the whole suite. Invoked through ${Python3_EXECUTABLE} rather than a bare
# `pytest`, so the suite runs under the interpreter the extension was built for.
set(MANTA_RUN_PYTEST "${CMAKE_CURRENT_BINARY_DIR}/run-pytest")
set(_manta_exports "")
foreach(_kv ${MANTA_PYTHON_ENV} "MANTA_PYTHON_ROOT=${MANTA_PYTHON_ROOT}")
  string(APPEND _manta_exports "export '${_kv}'\n")
endforeach()
file(GENERATE OUTPUT "${MANTA_RUN_PYTEST}"
  CONTENT "#!/bin/sh
# Generated by CMake. pytest over python/Tests against the build at
# ${PROJECT_BINARY_DIR}, as CTest runs it; arguments are pytest's.
${_manta_exports}[ $# -eq 0 ] && set -- '${CMAKE_CURRENT_SOURCE_DIR}'
exec '${Python3_EXECUTABLE}' -m pytest \\
  -c '${PROJECT_SOURCE_DIR}/pytest.ini' \\
  --rootdir '${PROJECT_SOURCE_DIR}' \\
  -o 'cache_dir=${CMAKE_CURRENT_BINARY_DIR}/pytest_cache' \\
  --basetemp '${CMAKE_CURRENT_BINARY_DIR}/tmp' \\
  \"$@\"
"
  FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE
                   WORLD_READ WORLD_EXECUTE)

add_test(NAME python
         COMMAND "${MANTA_RUN_PYTEST}"
         WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(python PROPERTIES LABELS "python")

add_custom_target(python_tests
  COMMAND "${CMAKE_CTEST_COMMAND}" --test-dir "${PROJECT_BINARY_DIR}" -R "^python$" --output-on-failure
  USES_TERMINAL
  COMMENT "Running the Python test suite")
add_dependencies(python_tests _manta)
